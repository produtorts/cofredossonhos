}

  const desde = Date.now() - 15 * 60e3;
  const r = await env.DB.prepare('SELECT COUNT(*) n FROM tentativas WHERE ip = ? AND quando > ?').bind(ip, desde).first();
  return (r?.n || 0) >= max;
}
async function anotarTentativa(env, ip) {
  await env.DB.batch([
    env.DB.prepare('INSERT INTO tentativas (ip, quando) VALUES (?, ?)').bind(ip, Date.now()),
    env.DB.prepare('DELETE FROM tentativas WHERE quando < ?').bind(Date.now() - 864e5),
  ]);
}

async function criarSessao(env, usuario) {
  const token = aleatorio();
  await env.DB.prepare('INSERT INTO sessoes (token, usuario, expira) VALUES (?, ?, ?)').bind(await sha(token), usuario, Date.now() + DIAS * 864e5).run();
  return fazCookie(COOKIE, token, DIAS);
}
async function quemEsta(env, req) {
  const t = cookie(req, COOKIE);
  if (!t || !/^[0-9a-f]{64}$/.test(t)) return null;
  const r = await env.DB.prepare('SELECT u.id, u.nome, u.email, u.whatsapp, u.visto FROM sessoes s JOIN usuarios u ON u.id = s.usuario WHERE s.token = ? AND s.expira > ?').bind(await sha(t), Date.now()).first();
  if (r && Date.now() - (r.visto || 0) > 10 * 60e3) await env.DB.prepare('UPDATE usuarios SET visto = ? WHERE id = ?').bind(Date.now(), r.id).run();
  return r;
}

// trabalho vindo do app: valida tudo (nomes, valores em centavos)
function limparTrabalho(b) {
  const titulo = String(b.titulo || '').trim().slice(0, 60) || 'Trabalho sem nome';
  const itens = (Array.isArray(b.itens) ? b.itens : []).slice(0, 2000).map((it) => ({
    id: String(it.id || '').replace(/[^a-z0-9]/gi, '').slice(0, 24) || aleatorio(6),
    nome: String(it.nome || '').trim().replace(/\s+/g, ' ').slice(0, 50),
    valor: Math.max(0, Math.min(99999999999, Math.round(Number(it.valor) || 0))),
    em: Number(it.em) || Date.now(),
  })).filter((it) => it.nome && it.valor > 0);
  const n = (x) => (Number.isFinite(+x) && +x > 0 ? Math.round(+x) : null);
  return { titulo, itens, criado: n(b.criado) || Date.now(), finalizado: n(b.finalizado) };
}

// ---------- painel do dono ----------
// segredo aleatório criado uma vez e guardado no banco (assina o cookie do painel)
async function tokenPainel(env) {
  let r = await env.DB.prepare("SELECT valor FROM config WHERE chave = 'segredo'").first();
  if (!r) { await env.DB.prepare("INSERT OR IGNORE INTO config (chave, valor) VALUES ('segredo', ?)").bind(aleatorio()).run(); r = await env.DB.prepare("SELECT valor FROM config WHERE chave = 'segredo'").first(); }
  return sha('painel|' + r.valor);
}
async function painelOk(env, req) { const t = cookie(req, COOKIE_PAINEL); return !!t && iguais(t, await tokenPainel(env)); }

export default {
  async fetch(req, env) {
    const url = new URL(req.url), p = url.pathname, m = req.method;
    try {
      if (p === '/' || p === '/index.html') return pagina(APP_HTML);
      if (p === '/painel' || p === '/painel/') return pagina(PAINEL_HTML);
      if (!p.startsWith('/sp/')) return Response.redirect(url.origin + '/', 302);
      if (m !== 'GET' && m !== 'HEAD') {
        const o = req.headers.get('origin');
        if (o && o !== url.origin) return erro('Origem não permitida.', 403);
      }
      await preparar(env);
      const corpo = m === 'POST' || m === 'PUT' ? await req.json().catch(() => ({})) : {};
      const ip = ipDe(req);

      // ---- contas ----
      if (p === '/sp/cadastro' && m === 'POST') {
        const nome = String(corpo.nome || '').trim().replace(/\s+/g, ' ').slice(0, 60);
        const email = String(corpo.email || '').trim().toLowerCase().slice(0, 120);
        const whatsapp = zap(corpo.whatsapp);
        const senha = String(corpo.senha || '');
        if (nome.length < 2) return erro('Escreva o seu nome.');
        if (!emailOk(email)) return erro('Confira o seu e-mail.');
        const certo = sugerirEmail(email);
        if (certo && !corpo.confirmouEmail) return json({ erro: 'Confira o e-mail: você quis dizer ' + certo + '?', sugestao: certo }, 400);
        if (!whatsapp) return erro('Coloque o WhatsApp com DDD, só números. Ex.: 11 91234-5678');
        if (senha.length < 6) return erro('A senha precisa ter pelo menos 6 caracteres.');
        if (await muitasTentativas(env, 'c:' + ip, 10)) return erro('Muitos cadastros seguidos daqui. Espere alguns minutos.', 429);
        await anotarTentativa(env, 'c:' + ip);
        const existe = await env.DB.prepare('SELECT id FROM usuarios WHERE email = ?').bind(email).first();
        if (existe) return erro('Esse e-mail já tem conta. Toque em "Já tenho conta" para entrar.', 409);
        const id = aleatorio(10), sal = aleatorio(16);
        await env.DB.prepare('INSERT INTO usuarios (id, nome, email, whatsapp, senha, sal, criado, visto) VALUES (?, ?, ?, ?, ?, ?, ?, ?)')
          .bind(id, nome, email, whatsapp, await hashSenha(senha, sal), sal, Date.now(), Date.now()).run();
        return json({ ok: true, nome }, 200, { 'set-cookie': await criarSessao(env, id) });
      }
      if (p === '/sp/entrar' && m === 'POST') {
        if (await muitasTentativas(env, ip)) return erro('Muitas tentativas erradas. Espere 15 minutos e tente de novo.', 429);
        const email = String(corpo.email || '').trim().toLowerCase();
        const u = await env.DB.prepare('SELECT id, nome, senha, sal FROM usuarios WHERE email = ?').bind(email).first();
        const ok = u && iguais(await hashSenha(String(corpo.senha || ''), u.sal), u.senha);
        if (!ok) { await anotarTentativa(env, ip); await new Promise((r) => setTimeout(r, 500)); return erro('E-mail ou senha incorretos.', 401); }
        return json({ ok: true, nome: u.nome }, 200, { 'set-cookie': await criarSessao(env, u.id) });
      }
      if (p === '/sp/sair' && m === 'POST') {
        const t = cookie(req, COOKIE);
        if (t) await env.DB.prepare('DELETE FROM sessoes WHERE token = ?').bind(await sha(t)).run();
        return json({ ok: true }, 200, { 'set-cookie': fazCookie(COOKIE, '', 0) });
      }

      // ---- painel do dono ----
      if (p === '/sp/painel/entrar' && m === 'POST') {
        let ok = false;
        try {
          const r = await fetch(CONFERE_SENHA, { method: 'POST', headers: { apikey: CHAVE_PUB, 'content-type': 'application/json' }, body: JSON.stringify({ senha: String(corpo.senha || ''), verificar: true, ip_cliente: 'somapix:' + ip }) });
          const d = await r.json().catch(() => ({}));
          if (r.status === 429) return erro(d.erro || 'Muitas tentativas. Espere alguns minutos.', 429);
          ok = r.ok && d.ok === true;
        } catch { return erro('Não consegui conferir a senha agora. Tente de novo.', 503); }
        if (!ok) return erro('Senha incorreta.', 401);
        return json({ ok: true }, 200, { 'set-cookie': fazCookie(COOKIE_PAINEL, await tokenPainel(env), 30) });
      }
      if (p.startsWith('/sp/painel/')) {
        if (!(await painelOk(env, req))) return erro('Entre com a senha do painel.', 401);
        if (p === '/sp/painel/sair') return json({ ok: true }, 200, { 'set-cookie': fazCookie(COOKIE_PAINEL, '', 0) });
        if (p === '/sp/painel/dados' && m === 'GET') {
          const { results: us } = await env.DB.prepare('SELECT id, nome, email, whatsapp, criado, visto FROM usuarios ORDER BY criado DESC LIMIT 1000').all();
          const { results: ts } = await env.DB.prepare('SELECT usuario, itens, finalizado FROM trabalhos').all();
          const por = {};
          for (const t of ts) {
            const x = (por[t.usuario] ||= { trabalhos: 0, finalizados: 0, pix: 0, total: 0 });
            const itens = JSON.parse(t.itens || '[]');
            x.trabalhos++; if (t.finalizado) x.finalizados++;
            x.pix += itens.length; x.total += itens.reduce((s, i) => s + (i.valor || 0), 0);
          }
          return json({ usuarios: us.map((u) => ({ ...u, ...(por[u.id] || { trabalhos: 0, finalizados: 0, pix: 0, total: 0 }) })) });
        }
        if (p === '/sp/painel/senha' && m === 'POST') {
          const nova = String(corpo.senha || '');
          if (nova.length < 6) return erro('A senha nova precisa ter pelo menos 6 caracteres.');
          const sal = aleatorio(16);
          const r = await env.DB.prepare('UPDATE usuarios SET senha = ?, sal = ? WHERE id = ?').bind(await hashSenha(nova, sal), sal, String(corpo.id || '')).run();
          if (!r.meta.changes) return erro('Conta não encontrada.', 404);
          await env.DB.prepare('DELETE FROM sessoes WHERE usuario = ?').bind(String(corpo.id || '')).run();
          return json({ ok: true });
        }
        return erro('Não encontrado.', 404);
      }

      // ---- daqui para baixo, precisa estar logado ----
      const eu = await quemEsta(env, req);
      if (!eu) return erro('Entre na sua conta.', 401);
      if (p === '/sp/eu' && m === 'GET') return json({ nome: eu.nome, email: eu.email });

      if (p === '/sp/trabalhos' && m === 'GET') {
        // mais recentes primeiro, de 30 em 30 ("Ver mais antigos" manda o cursor do último que chegou)
        const limite = Math.min(100, Math.max(1, parseInt(url.searchParams.get('limite') || '30', 10) || 30));
        const antes = Number(url.searchParams.get('antes')) || 0, antesId = String(url.searchParams.get('id') || '');
        const { results } = antes
          ? await env.DB.prepare('SELECT id, titulo, itens, criado, atualizado, finalizado FROM trabalhos WHERE usuario = ? AND (atualizado < ? OR (atualizado = ? AND id < ?)) ORDER BY atualizado DESC, id DESC LIMIT ?').bind(eu.id, antes, antes, antesId, limite + 1).all()
          : await env.DB.prepare('SELECT id, titulo, itens, criado, atualizado, finalizado FROM trabalhos WHERE usuario = ? ORDER BY atualizado DESC, id DESC LIMIT ?').bind(eu.id, limite + 1).all();
        const mais = results.length > limite, lista = results.slice(0, limite), ult = lista[lista.length - 1];
        return json({ nome: eu.nome, trabalhos: lista.map((t) => ({ ...t, itens: JSON.parse(t.itens || '[]') })), mais, cursor: mais && ult ? { antes: ult.atualizado, id: ult.id } : null });
      }
      // relatório: todos os Pix lançados entre ini e fim (em milissegundos), de todos os trabalhos
      if (p === '/sp/pix' && m === 'GET') {
        const ini = Number(url.searchParams.get('ini')) || 0, fim = Number(url.searchParams.get('fim')) || 0;
        if (!ini || !fim || fim < ini || fim - ini > 100 * 864e5) return erro('Período inválido.');
        const folga = 2 * 864e5; // relógio do celular pode estar um pouco diferente do servidor
        const { results } = await env.DB.prepare('SELECT id, titulo, itens, criado FROM trabalhos WHERE usuario = ? AND criado < ? AND atualizado >= ?').bind(eu.id, fim + folga, ini - folga).all();
        const pix = [];
        for (const t of results) for (const it of JSON.parse(t.itens || '[]')) {
          const em = it.em || t.criado;
          if (em >= ini && em < fim) pix.push({ id: it.id, nome: it.nome, valor: it.valor, em, trabalho: t.titulo, trabalhoId: t.id });
        }
        pix.sort((a, b) => a.em - b.em);
        return json({ pix });
      }
      const mt = p.match(/^\/sp\/trabalhos\/([a-z0-9]{6,24})$/i);
      if (mt && m === 'PUT') {
        const t = limparTrabalho(corpo), agora = Date.now();
        const dono = await env.DB.prepare('SELECT usuario FROM trabalhos WHERE id = ?').bind(mt[1]).first();
        if (dono && dono.usuario !== eu.id) return erro('Não encontrado.', 404);
        await env.DB.prepare(`INSERT INTO trabalhos (id, usuario, titulo, itens, criado, atualizado, finalizado) VALUES (?, ?, ?, ?, ?, ?, ?)
          ON CONFLICT(id) DO UPDATE SET titulo = excluded.titulo, itens = excluded.itens, atualizado = excluded.atualizado, finalizado = excluded.finalizado`)
          .bind(mt[1], eu.id, t.titulo, JSON.stringify(t.itens), t.criado, agora, t.finalizado).run();
        return json({ ok: true, atualizado: agora });
      }
      if (mt && m === 'DELETE') {
        await env.DB.prepare('DELETE FROM trabalhos WHERE id = ? AND usuario = ?').bind(mt[1], eu.id).run();
        return json({ ok: true });
      }
      return erro('Não encontrado.', 404);
    } catch (e) {
      console.error(e);
      return erro('Tive um problema agora. Tente de novo em instantes.', 500);
    }
  },
};
